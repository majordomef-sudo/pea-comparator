// ── In-memory localStorage replacement ──────────────────────────────────────
// Kakeibo never persists data to the real browser localStorage: everything
// lives in memory for the session and in Supabase as the durable copy.
// Must run before any code reads/writes localStorage at parse time.
(function(){
  // Keep a handle on the real, persistent localStorage before shimming it,
  // so the Supabase auth session can survive browser restarts even though
  // app data itself stays in-memory only.
  try { window.__kkRealLocalStorage = window.localStorage; } catch(e) {}
  const map = new Map();
  const store = {
    getItem: k => map.has(k) ? map.get(k) : null,
    setItem: (k,v) => { map.set(k, String(v)); },
    removeItem: k => { map.delete(k); },
    clear: () => { map.clear(); },
    key: i => Array.from(map.keys())[i] ?? null,
    get length(){ return map.size; }
  };
  try {
    Object.defineProperty(window, 'localStorage', { value: store, configurable: true });
  } catch(e) {
    console.error('Kakeibo: could not override localStorage', e);
  }
})();
</script>

// Dismiss splash once the app is actually ready (data pulled and rendered) —
// boot() dispatches 'kk:appReady' for this. This listener (and the 8s hard
// fallback) MUST be registered this early, before any blocking third-party
// script (e.g. the Chart.js CDN tag further down the page): boot() can
// resolve and dispatch 'kk:appReady' before a later <script> tag has even
// finished loading, and an event dispatched with no listener yet attached
// is simply lost — the splash would then sit stuck until whatever else
// eventually ran, which is what an 8s "stuck" load actually was.
let _splashDismissed=false;
function dismissSplash(fast){
  if(_splashDismissed)return;
  _splashDismissed=true;
  const s=document.getElementById('kkSplash');
  if(!s)return;
  if(fast){ s.remove(); return; }
  s.classList.add('hide');setTimeout(()=>s.remove(),550);
}
